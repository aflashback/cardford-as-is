import assert from 'node:assert/strict';
import {writeFileSync} from 'node:fs';
import pg from 'pg';

const url = new URL(process.env.TEST_DATABASE_URL);
assert.equal(process.platform, 'win32');
assert.equal(url.hostname, '127.0.0.1');
assert.equal(url.port, '54331');
assert.equal(url.pathname, '/cardford_windows_test');
const client = new pg.Client({connectionString:url.href, connectionTimeoutMillis:3000,
  query_timeout:5000, application_name:'cardford:m04d-cleanup-inspection'});
try {
  await client.connect();
  assert.equal((await client.query('SELECT current_database() AS name')).rows[0].name, 'cardford_windows_test');
  const activity = (await client.query(`SELECT datname, pid, application_name, backend_start, state_change, state,
    client_addr, client_port, pg_blocking_pids(pid) AS blockers
    FROM pg_stat_activity WHERE backend_type='client backend' AND pid<>pg_backend_pid() ORDER BY pid`)).rows;
  const isolatedSchemas = (await client.query(`SELECT nspname FROM pg_namespace
    WHERE left(nspname,4)='m03_' OR nspname='cardford_go_local' ORDER BY nspname`)).rows.map(x=>x.nspname);
  const proof = [];
  for (const schema of isolatedSchemas) {
    assert.match(schema, /^(m03_[a-z0-9_]+|cardford_go_local)$/);
    const tables = (await client.query('SELECT tablename FROM pg_tables WHERE schemaname=$1 ORDER BY tablename',[schema])).rows.map(x=>x.tablename);
    const migrations = tables.includes('schema_migrations')
      ? (await client.query(`SELECT name, applied_at FROM "${schema}".schema_migrations ORDER BY name`)).rows : [];
    proof.push({schema,tables,migrations});
  }
  const result={checkedAt:new Date().toISOString(),activity,isolatedSchemas,proof};
  writeFileSync('qa/implementation/multi-game/m04d/database-after-tests.json',JSON.stringify(result,null,2)+'\n');
  console.log(JSON.stringify(result));
} catch(error) {
  console.error('Database snapshot failed:',error.code ?? error.name);
  process.exitCode=1;
} finally {
  await client.end();
}
