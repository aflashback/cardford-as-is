import pg from 'pg';
import { writeFileSync } from 'node:fs';
const url = new URL(process.env.TEST_DATABASE_URL);
if(url.hostname!=='127.0.0.1'||url.port!=='54331'||url.pathname!=='/cardford_windows_test')throw Error('Unexpected database');
const client=new pg.Client({connectionString:url.href,connectionTimeoutMillis:5000,query_timeout:5000});
const schemas=['m03_service_299d5e229f432a36','m03_service_b2b7b9c7e655e158','m03_store_0ce949bd91a79c6d'];
try {
 await client.connect();
 const activity=(await client.query("SELECT pid,application_name,backend_start,state_change,state,client_addr,client_port,pg_blocking_pids(pid) AS blockers FROM pg_stat_activity WHERE datname='cardford_windows_test' AND pid<>pg_backend_pid() ORDER BY pid")).rows;
 const proof=[];
 for(const schema of schemas) {
  const tables=(await client.query('SELECT tablename FROM pg_tables WHERE schemaname=$1 ORDER BY tablename',[schema])).rows.map(r=>r.tablename);
  const migrations=tables.includes('schema_migrations')?(await client.query(`SELECT name,applied_at FROM "${schema}".schema_migrations ORDER BY name`)).rows:[];
  const games=tables.includes('m03_games')?(await client.query(`SELECT game_id,revision FROM "${schema}".m03_games LIMIT 10`)).rows:[];
  proof.push({schema,tables,migrations,games});
 }
 const result={checkedAt:new Date().toISOString(),activity,proof};
 writeFileSync('qa/implementation/multi-game/m04c/database-cleanup-proof.json',JSON.stringify(result,null,2)+'\n');
 console.log(JSON.stringify(result));
} finally {await client.end();}
