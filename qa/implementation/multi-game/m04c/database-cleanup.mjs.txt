import pg from 'pg';
import assert from 'node:assert/strict';
import { readFileSync, writeFileSync } from 'node:fs';
const url = new URL(process.env.TEST_DATABASE_URL);
assert.equal(url.hostname, '127.0.0.1'); assert.equal(url.port, '54331'); assert.equal(url.pathname, '/cardford_windows_test');
const proof = JSON.parse(readFileSync('qa/implementation/multi-game/m04c/database-cleanup-proof.json'));
const schemas = ['m03_service_299d5e229f432a36','m03_service_b2b7b9c7e655e158','m03_store_0ce949bd91a79c6d'];
assert.deepEqual(proof.proof.map(p=>p.schema), schemas);
for (const p of proof.proof) {
  assert.ok(p.migrations.every(m=>m.applied_at >= '2026-10-06T19:37:21.000Z'));
  assert.ok(p.games.every(g=>g.game_id==='m04-cancellation' && g.revision==='0'));
}
const client = new pg.Client({ connectionString:url.href, connectionTimeoutMillis:5000, query_timeout:5000 });
try {
 await client.connect();
 const result={at:new Date().toISOString(),database:url.pathname.slice(1),dropped:[],terminatedStaleTestBackend:false};
 await client.query('BEGIN');
 for (const schema of schemas) {
  const active=await client.query('SELECT 1 FROM pg_stat_activity WHERE application_name=$1 AND pid<>pg_backend_pid()',['cardford:'+schema]);
  assert.equal(active.rowCount,0,'Fixture still has a client');
  await client.query(`DROP SCHEMA "${schema}" CASCADE`); result.dropped.push(schema);
 }
 await client.query('COMMIT');
 // This task's idle canceled-test backend has no remaining OS TCP client or
 // Go/test process (checked separately). Its schema has already been removed.
 const stale=await client.query("SELECT pg_terminate_backend(pid) AS terminated FROM pg_stat_activity WHERE pid=22204 AND datname='cardford_windows_test' AND application_name='cardford:m03_store_0c30546d1873c1f1' AND state='idle' AND backend_start >= '2026-10-06T19:45:41Z' AND backend_start < '2026-10-06T19:45:42Z' AND cardinality(pg_blocking_pids(pid))=0");
 result.terminatedStaleTestBackend=stale.rows[0]?.terminated===true;
 writeFileSync('qa/implementation/multi-game/m04c/database-cleanup.json',JSON.stringify(result,null,2)+'\n'); console.log(JSON.stringify(result));
} finally { await client.end(); }
