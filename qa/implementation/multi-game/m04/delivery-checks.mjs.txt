import { readFileSync, readdirSync, existsSync, writeFileSync } from 'node:fs';
import { createHash } from 'node:crypto';
import { spawnSync } from 'node:child_process';
import { resolve, dirname } from 'node:path';
const root = process.cwd(), folder = 'qa/implementation/multi-game/m04';
const names = ['go-full','go-race','go-vet','go-contracts','public-contracts','oracle-check','node-reference'];
const sha = value => createHash('sha256').update(value).digest('hex');
const reports = names.map(name => JSON.parse(readFileSync(`${folder}/${name}.json`)));
for (const report of reports) {
  if (report.exitCode !== 0 || !report.sourceUnchanged) throw Error(`Failed acceptance ${report.name}`);
  if (sha(readFileSync(`${folder}/${report.output}`)) !== report.outputSHA256) throw Error(`Output drift ${report.name}`);
  for (const file of report.source) if (sha(readFileSync(file.path)) !== file.sha256) throw Error(`Source drift ${file.path}`);
}
if (new Set(reports.map(r=>r.sourceSHA256)).size !== 1) throw Error('Final source manifests differ');
const preserved = ['web', 'package.json', 'package-lock.json', 'go.mod', 'go.sum', 'scripts',
  'server/internal/games/spades', 'server/internal/contracts', 'server/internal/gamekernel',
  'qa/experiments', ...['m02','m02a','m03','performance-windows'].map(p=>`qa/implementation/multi-game/${p}`)];
const diff = spawnSync('git',['diff','--exit-code','dc0e44a','--',...preserved],{encoding:'utf8',windowsHide:true});
if (diff.status !== 0) throw Error('Preserved reference/runtime/evidence paths have a diff');
const docs = ['CHANGELOG.md','planning/CHANGELOG.md','planning/IMPLEMENTATION-START-HERE.md','planning/README.md',
 'planning/implementation/multi-game/m04-execution.md','planning/implementation/multi-game/backend-readiness.md',
 'qa/implementation/multi-game/status.md','docs/superpowers/plans/2026-10-04-multi-game-reasoning-implementation.md',
 ...readdirSync(folder).filter(p=>p.endsWith('.md')).map(p=>`${folder}/${p}`)];
let links = 0;
for (const file of docs) for (const match of readFileSync(file,'utf8').matchAll(/\[[^\]]*\]\(([^)]+)\)/g)) {
  const href = match[1].split('#')[0];
  if (!href || /^[a-z]+:/.test(href)) continue;
  if (!existsSync(resolve(dirname(file), decodeURIComponent(href)))) throw Error(`Broken link ${file} -> ${href}`);
  links++;
}
const credentials = JSON.parse(readFileSync('.local/windows-postgres/credentials.json'));
for (const file of readdirSync(folder)) {
  const data = readFileSync(`${folder}/${file}`,'utf8');
  if (data.includes(credentials.password)) throw Error(`Credential in evidence ${file}`);
}
const result = { checkedAt:new Date().toISOString(), checks:names, sourceSHA256:reports[0].sourceSHA256,
  sourceAndOutputHashes:'match final captured bytes', preservedFrom:'dc0e44a', preservedPaths:preserved,
  localMarkdownLinks:links, privateDatabaseCredential:'absent from M04 evidence' };
writeFileSync(`${folder}/delivery-checks.json`, JSON.stringify(result,null,2)+'\n');
console.log(JSON.stringify(result));
